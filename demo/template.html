Con JavaScript

      
import { create } from 'pocketchart';

const chart = create('#grafico', {
  type: 'area',
  title: 'Consumo',
  animation: { enter: 'grow' },
});
chart.load(() => fetch('/api/consumo').then((r) => r.json()));
chart.on('inspect', (sel) => console.log(sel));
